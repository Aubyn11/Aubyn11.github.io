39. 组合总数

难度：中等
题目要求：
给定一个无重复元素的整数数组candidates和一个目标整数target，找出candidates中可以使数字和为目标数target所有不同组合，并以列表形式返回。你可以按任意顺序返回这些组合。

candidates的同一个可以无限制重复被选取。如果至少一个数字的被选数量不同，则两种组合是不同的。

对于给定的输入，保证和为target的不同组合数少于150个。

示例1：

输入：candidates = [2,3,6,7], target = 7
输出：[[2,2,3],[7]]

示例2：

输入：candidates = [2,3,5], target = 8
输出：[[2,2,2,2],[2,3,3],[3,5]]

示例3：

输入：candidates = [2], target = 1
输出：[]

题解

直接利用回溯法，先用一个数取尽，再取下一个数，如果下一个数无法取得，就取出一个数继续尝试，直到最后的值减到0

想法代码：

class Solution
{
    public List<IList<int>> res = new List<IList<int>>();
    public List<int> list = new List<int>();

    public static void Main(String[] args)
    {
        int[] candidates = { 2, 3, 6, 7 };
        int target = 7;
        Solution solution = new Solution();
        IList<IList<int>> result = solution.CombinationSum(candidates, target);
        foreach (var a in result)
        {
            foreach (var b in a)
            {
                Console.Write(b + " ");
            }
            Console.WriteLine();
        }
    }

    public IList<IList<int>> CombinationSum(int[] candidates, int target)
    {
        Recall(candidates, target, 0);
        return res;
    }

    public void Recall(int[] cadidates, int target, int start)
    {
        if (target < 0)
        {
            return;
        }

        if (target == 0)
        {
            res.Add(new List<int>(list));
            return;
        }

        for (int i = start; i < cadidates.Length; i++)
        {
            list.Add(cadidates[i]);
            Recall(cadidates,target - cadidates[i], start);
            list.RemoveAt(list.Count - 1);
            start++;
        }
    }
}

40. 组合总数 II

难度：中等
题目要求：
给定一个候选人编号的集合candidates和一个目标数target，找出candidates中所有可以使数字和为target的组合。

candidates中的每个数字在每个组合中只能使用一次。

示例1：

输入：candidates = [10,1,2,7,6,1,5], target = 8
输出：
[
[1,1,6],
[1,2,5],
[1,7],
[2,6]
]

示例2：

输入：candidates = [2,5,2,1,2], target = 5
输出：
[
[1,2,2],
[5]
]

题解：

使用回溯法，先声名一个List数组存放dandidates中的不同数字的val和出现的次数，当$target>0$时，用target除以当前pos位置的curList[pos][0]的val，比较获取应该减去的次数，开始循环，$pos+1，target-curList[pos][0]，reList.Add(curList[pos][0]$，通过这个list数组记录val和出现次数

想法代码：

class Solution
{
    IList<IList<int>> res = new List<IList<int>>();
    IList<int> list = new List<int>();
    IList<int[]> curList = new List<int[]>();

    public static void Main(String[] args)
    {
        int[] candidates = { 2, 5, 2, 1, 2 };
        int target = 5;
        Solution solution = new Solution();
        IList<IList<int>> result = solution.CombinationSum2(candidates, target);
        foreach (var a in result)
        {
            foreach (var b in a)
            {
                Console.Write(b + " ");
            }
            Console.WriteLine();
        }
    }

    public IList<IList<int>> CombinationSum2(int[] candidates, int target)
    {
        Array.Sort(candidates);
        int len = candidates.Length;
        for (int i = 0; i < len; i++)
        {
            if (curList.Count == 0 || curList[curList.Count - 1][0] != candidates[i])
            {
                curList.Add(new int[] { candidates[i], 1 });
            }
            else
            {
                ++curList[curList.Count - 1][1];
            }
        }

        Recall(target, 0);
        return res;
    }

    public void Recall(int target, int pos)
    {
        if (target == 0)
        {
            res.Add(new List<int>(list));
            return;
        }
        if (pos >= curList.Count || target < curList[pos][0])
        {
            return;
        }
        Recall(target, pos + 1);
        int min = Math.Min(target / curList[pos][0], curList[pos][1]);
        for (int i = 1; i <= min; i++)
        {
            list.Add(curList[pos][0]);
            Recall(target - i * curList[pos][0], pos + 1);
        }

        for (int i = 1; i <= min; i++)
        {
            list.RemoveAt(list.Count - 1);
        }
    }
}
